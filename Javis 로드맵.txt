### Step 1-2. LLM 호출 래퍼 (`core/llm.py`)

```python
import os
import anthropic
from dotenv import load_dotenv

load_dotenv()

client = anthropic.Anthropic(api_key=os.getenv("ANTHROPIC_API_KEY"))

SYSTEM_PROMPT = """
당신은 지원의 개인 AI 비서입니다.
- 간결하고 실용적으로 답변하세요.
- 모르는 것은 솔직하게 모른다고 하세요.
- 한국어로 대화하세요.
"""

def chat(messages: list[dict]) -> str:
    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=4096,
        system=SYSTEM_PROMPT,
        messages=messages
    )
    return response.content[0].text
```

### Step 1-3. 대화 히스토리 저장 (`core/memory.py`)

```python
import sqlite3
import json
from datetime import datetime

DB_PATH = "assistant.db"

def init_db():
    conn = sqlite3.connect(DB_PATH)
    conn.execute("""
        CREATE TABLE IF NOT EXISTS conversations (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            session_id TEXT,
            role TEXT,
            content TEXT,
            timestamp TEXT
        )
    """)
    conn.commit()
    conn.close()

def save_message(session_id: str, role: str, content: str):
    conn = sqlite3.connect(DB_PATH)
    conn.execute(
        "INSERT INTO conversations VALUES (NULL, ?, ?, ?, ?)",
        (session_id, role, content, datetime.now().isoformat())
    )
    conn.commit()
    conn.close()

def load_session(session_id: str) -> list[dict]:
    conn = sqlite3.connect(DB_PATH)
    rows = conn.execute(
        "SELECT role, content FROM conversations WHERE session_id=? ORDER BY id",
        (session_id,)
    ).fetchall()
    conn.close()
    return [{"role": r[0], "content": r[1]} for r in rows]
```

### Step 1-4. CLI 메인 루프 (`main.py`)

```python
import uuid
from core.llm import chat
from core.memory import init_db, save_message, load_session

def main():
    init_db()
    session_id = str(uuid.uuid4())
    messages = []

    print("🤖 AI 비서 시작. 종료: 'exit' 입력\n")

    while True:
        try:
            user_input = input("나: ").strip()
        except (KeyboardInterrupt, EOFError):
            print("\n종료합니다.")
            break

        if user_input.lower() in ("exit", "quit", "종료"):
            break
        if not user_input:
            continue

        messages.append({"role": "user", "content": user_input})
        save_message(session_id, "user", user_input)

        response = chat(messages)
        messages.append({"role": "assistant", "content": response})
        save_message(session_id, "assistant", response)

        print(f"\n비서: {response}\n")

if __name__ == "__main__":
    main()
```

### Step 2-1. Tool Use 개념 이해

Claude의 Tool Use는 3단계로 동작합니다.

```
1. 사용자 메시지 전송
        ↓
2. Claude가 "이 툴을 써야겠다" 판단 → tool_use 블록 반환
        ↓
3. 우리 코드가 실제로 툴 실행 → 결과를 tool_result로 전달
        ↓
4. Claude가 결과를 보고 최종 답변 생성
```

```python
        "description": "웹에서 최신 정보를 검색합니다.",

                "query": {"type": "string", "description": "검색어"}

# 실제 도구 구현
def read_file(path: str) -> str:
    try:
        with open(os.path.expanduser(path), "r", encoding="utf-8") as f:
            return f.read()
    except Exception as e:
        return f"오류: {e}"

def write_file(path: str, content: str) -> str:
    try:
        with open(os.path.expanduser(path), "w", encoding="utf-8") as f:
            f.write(content)
        return f"✓ {path} 저장 완료"
    except Exception as e:
        return f"오류: {e}"

def run_command(command: str) -> str:
    try:
        result = subprocess.run(
            command, shell=True, capture_output=True,
            text=True, timeout=30
        )
        output = result.stdout + result.stderr
        return output[:3000] if output else "(출력 없음)"
    except subprocess.TimeoutExpired:
        return "오류: 30초 초과"
    except Exception as e:
        return f"오류: {e}"

def web_search(query: str) -> str:
    try:
        from duckduckgo_search import DDGS
        with DDGS() as ddgs:
            results = list(ddgs.text(query, max_results=5))
        return "\n\n".join(
            f"제목: {r['title']}\n내용: {r['body']}\nURL: {r['href']}"
            for r in results
        )
    except Exception as e:
        return f"검색 실패: {e}"

# 도구 실행 디스패처
TOOL_HANDLERS = {
    "read_file": lambda inp: read_file(inp["path"]),
    "write_file": lambda inp: write_file(inp["path"], inp["content"]),
    "run_command": lambda inp: run_command(inp["command"]),
    "web_search": lambda inp: web_search(inp["query"]),
}

def execute_tool(tool_name: str, tool_input: dict) -> str:
    handler = TOOL_HANDLERS.get(tool_name)
    if not handler:
        return f"알 수 없는 도구: {tool_name}"
    return handler(tool_input)
```

### Step 2-3. Tool Use 루프 (`core/llm.py` 업데이트)

```python
from core.tools import TOOL_SPECS, execute_tool

def chat_with_tools(messages: list[dict]) -> str:
    while True:
        response = client.messages.create(
            model="claude-sonnet-4-6",
            max_tokens=4096,
            system=SYSTEM_PROMPT,
            tools=TOOL_SPECS,
            messages=messages
        )

        # 도구 사용 없이 바로 텍스트 응답
        if response.stop_reason == "end_turn":
            text_blocks = [b.text for b in response.content if b.type == "text"]
            return "\n".join(text_blocks)

        # 도구 사용 요청 처리
        if response.stop_reason == "tool_use":
            # Claude의 응답 메시지 추가
            messages.append({"role": "assistant", "content": response.content})

            # 각 도구 실행 후 결과 수집
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    print(f"  🔧 [{block.name}] 실행 중...")
                    result = execute_tool(block.name, block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": result
                    })

            # 결과를 user 메시지로 전달하고 루프 계속
            messages.append({"role": "user", "content": tool_results})
```

```python
import json
import os

PROFILE_PATH = "user_profile.json"

DEFAULT_PROFILE = {
    "name": "",
    "preferences": {},
    "facts": [],        # 기억할 사실들: ["서울 거주", "Python 개발자"]
    "shortcuts": {}     # 커스텀 단축 명령어
}

def load_profile() -> dict:
    if os.path.exists(PROFILE_PATH):
        with open(PROFILE_PATH) as f:
            return json.load(f)
    return DEFAULT_PROFILE.copy()

def save_profile(profile: dict):
    with open(PROFILE_PATH, "w", encoding="utf-8") as f:
        json.dump(profile, f, ensure_ascii=False, indent=2)

def build_profile_context(profile: dict) -> str:
    """시스템 프롬프트에 주입할 사용자 정보"""
    if not profile.get("name") and not profile.get("facts"):
        return ""
    lines = ["## 사용자 정보"]
    if profile.get("name"):
        lines.append(f"- 이름: {profile['name']}")
    for fact in profile.get("facts", []):
        lines.append(f"- {fact}")
    return "\n".join(lines)
```

### Step 3-1. RAG 파이프라인 (`core/rag.py`)

RAG(Retrieval-Augmented Generation)는 "내 문서를 AI의 기억으로 만드는" 기술입니다.

```
문서 추가 시:
  PDF/TXT 파일 → 청크 분할 → 임베딩 벡터 생성 → 벡터 DB 저장

질문 시:
  질문 → 임베딩 → 유사 청크 검색 → 컨텍스트에 주입 → LLM 답변
```

```python
import chromadb
from sentence_transformers import SentenceTransformer
import re

# 임베딩 모델 (한국어 지원)
embedder = SentenceTransformer("jhgan/ko-sroberta-multitask")
chroma_client = chromadb.PersistentClient(path="./chroma_db")
collection = chroma_client.get_or_create_collection("my_docs")

def chunk_text(text: str, chunk_size=500, overlap=50) -> list[str]:
    """텍스트를 청크로 분할"""
    sentences = re.split(r'(?<=[.!?])\s+', text)
    chunks, current = [], ""
    for sent in sentences:
        if len(current) + len(sent) > chunk_size:
            if current:
                chunks.append(current.strip())
            current = current[-overlap:] + sent
        else:
            current += " " + sent
    if current:
        chunks.append(current.strip())
    return chunks

def add_document(file_path: str, doc_id: str = None):
    """문서를 RAG DB에 추가"""
    with open(file_path, encoding="utf-8") as f:
        text = f.read()

    chunks = chunk_text(text)
    embeddings = embedder.encode(chunks).tolist()
    doc_id = doc_id or file_path

    collection.add(
        documents=chunks,
        embeddings=embeddings,
        ids=[f"{doc_id}_chunk_{i}" for i in range(len(chunks))],
        metadatas=[{"source": file_path, "chunk": i} for i in range(len(chunks))]
    )
    print(f"✓ {len(chunks)}개 청크 저장 완료")

def search_docs(query: str, top_k=3) -> str:
    """관련 문서 청크 검색"""
    query_emb = embedder.encode([query]).tolist()
    results = collection.query(query_embeddings=query_emb, n_results=top_k)

    if not results["documents"][0]:
        return ""

    context_parts = []
    for doc, meta in zip(results["documents"][0], results["metadatas"][0]):
        context_parts.append(f"[출처: {meta['source']}]\n{doc}")

    return "\n\n---\n\n".join(context_parts)
```

```python
import subprocess
import tempfile
import os

def listen() -> str:
    """마이크에서 음성을 녹음하고 텍스트로 변환 (Whisper 사용)"""
    import whisper
    import sounddevice as sd
    import numpy as np
    import scipy.io.wavfile as wav

    print("🎤 말씀하세요... (3초)")
    sample_rate = 16000
    recording = sd.rec(
        int(3 * sample_rate), samplerate=sample_rate,
        channels=1, dtype='int16'
    )
    sd.wait()

    with tempfile.NamedTemporaryFile(suffix=".wav", delete=False) as f:
        wav.write(f.name, sample_rate, recording)
        model = whisper.load_model("base")
        result = model.transcribe(f.name, language="ko")
        os.unlink(f.name)
        return result["text"].strip()

def speak(text: str):
    """텍스트를 음성으로 출력"""
    import pyttsx3
    engine = pyttsx3.init()
    engine.setProperty('rate', 180)
    engine.say(text)
    engine.runAndWait()
```

```python
def run_scheduler():
    """백그라운드 스레드로 스케줄러 실행"""
    def loop():
        while True:
            schedule.run_pending()
            time.sleep(30)
    thread = threading.Thread(target=loop, daemon=True)
    thread.start()
```

### Step 4-1. 에이전트 루프 설계

ReAct(Reason + Act) 패턴으로 복잡한 작업을 단계별로 해결합니다.

```
[사용자 목표]
    ↓
[Thought: 어떻게 할까 생각]
    ↓
[Action: 도구 실행]
    ↓
[Observation: 결과 확인]
    ↓ (목표 달성될 때까지 반복)
[Final Answer]
```

```python
# core/agent.py

AGENT_SYSTEM = """
당신은 자율적으로 작업을 계획하고 실행하는 AI 에이전트입니다.

복잡한 작업을 받으면:
1. 작업을 작은 단계로 분해하세요.
2. 각 단계에 필요한 도구를 선택하세요.
3. 실행 결과를 보고 다음 단계를 결정하세요.
4. 최종 결과를 명확히 보고하세요.

불확실하면 사용자에게 확인을 요청하세요.
"""

class Agent:
    def __init__(self, max_iterations=10):
        self.max_iterations = max_iterations

    def run(self, goal: str) -> str:
        messages = [{"role": "user", "content": f"목표: {goal}"}]
        iteration = 0

        while iteration < self.max_iterations:
            iteration += 1
            print(f"\n[에이전트 루프 {iteration}/{self.max_iterations}]")

            response = client.messages.create(
                model="claude-sonnet-4-6",
                max_tokens=8096,
                system=AGENT_SYSTEM,
                tools=TOOL_SPECS,
                messages=messages
            )

            if response.stop_reason == "end_turn":
                final = "\n".join(
                    b.text for b in response.content if b.type == "text"
                )
                return final

            if response.stop_reason == "tool_use":
                messages.append({"role": "assistant", "content": response.content})
                results = []
                for block in response.content:
                    if block.type == "tool_use":
                        print(f"  🔧 {block.name}({block.input})")
                        result = execute_tool(block.name, block.input)
                        results.append({
                            "type": "tool_result",
                            "tool_use_id": block.id,
                            "content": result
                        })
                messages.append({"role": "user", "content": results})

        return "최대 반복 횟수 초과. 작업을 분할해서 시도해주세요."
```

```python
from mcp.server import Server
from mcp.server.stdio import stdio_server

server = Server("my-assistant-tools")

@server.list_tools()
async def list_tools():

                "type": "object",

                "required": ["city"]

        )
    ]

        return [types.TextContent(type="text", text=f"{city} 현재 22°C, 맑음")]

        await server.run(read, write, server.create_initialization_options())
```

### Step 4-3. 멀티모달 처리 (`core/vision.py`)

```python
import base64
from pathlib import Path

def encode_image(image_path: str) -> tuple[str, str]:
    """이미지를 base64로 인코딩"""
    path = Path(image_path)
    suffix_to_media = {
        ".jpg": "image/jpeg", ".jpeg": "image/jpeg",
        ".png": "image/png", ".gif": "image/gif", ".webp": "image/webp"
    }
    media_type = suffix_to_media.get(path.suffix.lower(), "image/jpeg")
    with open(path, "rb") as f:
        return base64.standard_b64encode(f.read()).decode("utf-8"), media_type

def chat_with_image(text: str, image_path: str) -> str:
    """이미지 + 텍스트 질문"""
    img_data, media_type = encode_image(image_path)
    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=2048,
        messages=[{
            "role": "user",
            "content": [
                {
                    "type": "image",
                    "source": {"type": "base64", "media_type": media_type, "data": img_data}
                },
                {"type": "text", "text": text}
            ]
        }]
    )
    return response.content[0].text
```
